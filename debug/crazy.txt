module VendingMachine {
  var : boolean;
  var max: integer;
  var nbeer: integer;
  var nsoda: integer;
  var sget: boolean;
  var bget: boolean;
  var refill: boolean;
  var coin: boolean;
  var ret_coin: boolean;

  init {
    nsoda = max;
    nbeer = max;
    sget = false;
    bget = false;
    refill = false;
    coin = true;
    ret_coin = false;

  }
  procedure step()
    modifies nbeer;
    modifies bget;
    modifies coin;
    modifies ret_coin;
    modifies sget;
    modifies nsoda;
  {
    if ((nsoda > 0) && (nbeer > 0)) {
      sget = true;
      bget = true;
    } else {
      sget = false;
      bget = false;
    }
    if ((nsoda == 0) && (nbeer == 0)) {
      ret_coin = true;
      coin = false;
    } else {
      ret_coin = false;
      coin = true;
    }
    if (sget) {
      nsoda = (nsoda - 1);
    }
    if (bget) {
      nbeer = (nbeer - 1);
    }
    if (refill) {
      nsoda = max;
      nbeer = max;
    }
  }

  next {
    call step();
  }
  invariant nsoda_inv : ((nsoda >= 0) && (nsoda <= max));
  invariant nbeer_inv : ((nbeer >= 0) && (nbeer <= max));
  invariant sget_inv : ((nsoda > 0) ==> sget);
  invariant bget_inv : ((nbeer > 0) ==> bget);
  invariant refill_inv : (((nsoda < max) || (nbeer < max)) ==> refill);
  invariant coin_inv : (coin == true);
  invariant ret_coin_inv : (((nsoda == 0) && (nbeer == 0)) ==> ret_coin);

  control  {
    v = bmc(3);
    check;
    print_results;
    v.print_cex();
  }
}
